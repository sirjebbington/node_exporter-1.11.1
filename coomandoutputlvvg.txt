#include <libperfstat.h>
#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <errno.h>

int main() {
    perfstat_config(PERFSTAT_ENABLE | PERFSTAT_LV | PERFSTAT_VG, NULL);

    /* ---- Volume Groups ---- */
    perfstat_id_t id;
    strcpy(id.name, "");
    errno = 0;
    int nvg = perfstat_volumegroup(NULL, NULL, sizeof(perfstat_volumegroup_t), 0);
    printf("=== VolumeGroupStat count call: n=%d errno=%d (%s) ===\n", nvg, errno, strerror(errno));

    if (nvg > 0) {
        perfstat_volumegroup_t *vg = malloc(sizeof(perfstat_volumegroup_t) * nvg);
        errno = 0;
        int rvg = perfstat_volumegroup(&id, vg, sizeof(perfstat_volumegroup_t), nvg);
        printf("=== VolumeGroupStat fetch call: r=%d errno=%d (%s) ===\n", rvg, errno, strerror(errno));
        for (int i = 0; i < rvg; i++) {
            printf("VG name=%-20s variedState=%u totalLVs=%llu openedLVs=%llu totalDisks=%llu activeDisks=%llu\n",
                   vg[i].name, vg[i].variedState, vg[i].total_logical_volumes, vg[i].opened_logical_volumes,
                   vg[i].total_disks, vg[i].active_disks);
        }
        free(vg);
    }

    printf("\n");

    /* ---- Logical Volumes ---- */
    strcpy(id.name, "");
    errno = 0;
    int nlv = perfstat_logicalvolume(NULL, NULL, sizeof(perfstat_logicalvolume_t), 0);
    printf("=== LogicalVolumeStat count call: n=%d errno=%d (%s) ===\n", nlv, errno, strerror(errno));

    if (nlv > 0) {
        perfstat_logicalvolume_t *lv = malloc(sizeof(perfstat_logicalvolume_t) * nlv);
        errno = 0;
        int rlv = perfstat_logicalvolume(&id, lv, sizeof(perfstat_logicalvolume_t), nlv);
        printf("=== LogicalVolumeStat fetch call: r=%d errno=%d (%s) ===\n", rlv, errno, strerror(errno));
        for (int i = 0; i < rlv; i++) {
            printf("LV name=%-20s vg=%-20s state=%llu open_close=%llu LPs=%llu ppsize=%llu mirrors=%u mirrorPolicy=%llu writeVerify=%llu version=%llu\n",
                   lv[i].name, lv[i].vgname, lv[i].state, lv[i].open_close, lv[i].logical_partitions,
                   lv[i].ppsize, lv[i].mirrors, lv[i].mirror_policy, lv[i].write_verify, lv[i].version);
        }
        free(lv);
    }

    return 0;
}
